    int largestRectangleArea(vector<int>& heights) {
      int n = heights.size(), maxi = 0;
      for (int i = 0; i < n; ++i) {
          int l = i, r = i;
          for (int j = i-1; j >= 0 && heights[i] <= heights[j]; --j) --l;
          for (int j = i+1; j < n && heights[i] <= heights[j]; ++j) ++r;
          maxi = max(maxi, (r-l+1) * heights[i]);
      }
      return maxi;
    }

    int largestRectangleArea(vector<int>& heights) {
      int maxi = 0, n = heights.size();
      for (int i = 0; i < n; ++i) {
          int minH = INT_MAX;
          for (int j = i; j < n; ++j) {
              minH = min(minH, heights[j]);
              maxi = max(maxi, minH * (j-i+1));
          }
      }
      return maxi;
    }

    int largestRectangleArea(vector<int>& heights) {
      int n = heights.size(), maxi = 0, i = 0;
      heights.push_back(0);
      stack<int> s;
      while(i <= n) {
          while(!s.empty() && heights[s.top()] > heights[i]) {
              int curr = s.top();
              s.pop();
              int h = heights[curr];
              if (s.empty())
                  maxi = max(maxi, h * i);
              else 
                  maxi = max(maxi, h * (i - s.top() - 1));
          }
          s.push(i);
          ++i;
      }
      return maxi;
    }
