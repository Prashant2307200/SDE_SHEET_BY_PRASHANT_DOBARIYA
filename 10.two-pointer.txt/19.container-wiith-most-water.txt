    int maxArea(vector<int>& height) {
        int l = 0, r = height.size() - 1;
        int maxi = 0;
        while(l < r) {
            int w = r - l;
            int h = min(height[r], height[l]);
            maxi = max(maxi, h * w);
            if(height[l] > height[r]) r--;
            else l++;
        }
        return maxi;
    }
